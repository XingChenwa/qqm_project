cmake_minimum_required(VERSION 3.15)
project(qqm_agent_js LANGUAGES C CXX)
if(NOT QJS_PACKAGE)
  message(FATAL_ERROR "Pass -DQJS_PACKAGE=<flutter_qjs_next package directory>")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include("${QJS_PACKAGE}/cxx/quickjs.cmake")
add_library(qqm_agent_js SHARED "${QJS_PACKAGE}/cxx/ffi.cpp")
target_link_libraries(qqm_agent_js PRIVATE quickjs)
if(MSVC)
  target_compile_features(qqm_agent_js PRIVATE cxx_std_20)
  target_compile_options(qqm_agent_js PRIVATE /utf-8 /wd4576 /wd4244)
else()
  target_link_libraries(qqm_agent_js PRIVATE m dl pthread)
endif()
